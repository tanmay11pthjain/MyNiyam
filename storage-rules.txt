rules_version = '2';

// MyNiyam v5 — Firebase Storage rules.
// Paste this into Firebase Console -> Storage -> Rules -> Publish.
// (If Storage has never been enabled on this project, click "Get started"
// first and accept the default bucket, then paste these rules.)
//
// One object per profile: profile-photos/{profileId}.jpg
// profileId is either the base Google/Firebase Auth uid (primary profile)
// or "{baseUid}__p2".."__p5" (a sibling profile on the same account) — see
// auth.js's PROFILE_SEP/MAX_PROFILES. Storage rules see the same request.auth.uid
// as Realtime Database rules do: always the BASE account, never the active
// profile slot — so ownership is checked the same way the database rules do.
//
// KNOWN LIMITATION, accepted deliberately: Storage security rules cannot
// read Realtime Database data (no request.auth custom claims and no RTDB
// `get()`/`exists()` — that cross-service read only exists for Firestore),
// so a per-sangh read check like the one in firebase-rules.json's
// users/$uid rule is not possible here without adding Cloud Functions +
// custom claims, which this project doesn't otherwise need. Read is
// therefore "any signed-in account", not "same-sangh admin only" — still a
// large improvement over today's actual exposure (the Apps Script
// get_photo endpoint returns any photo to anyone on the public internet
// who knows a uid, no sign-in required at all — see SANGH-RUNBOOK.md and
// the plan's Non-goals). If per-sangh photo privacy becomes a requirement,
// revisit with Cloud Functions issuing short-lived signed URLs instead.
// NOTE on the path pattern: a wildcard must be an ENTIRE path segment —
// `match /profile-photos/{profileId}.jpg` is invalid syntax and silently
// matches nothing (which denies every upload). So the whole filename is
// captured as one segment and compared against "{uid}.jpg" directly.
service firebase.storage {
  match /b/{bucket}/o {
    match /profile-photos/{fileName} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                   && (fileName == request.auth.uid + '.jpg'
                       || fileName.matches(request.auth.uid + '__p[2-5][.]jpg'))
                   && request.resource.size < 2 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
      allow delete: if request.auth != null
                    && (fileName == request.auth.uid + '.jpg'
                        || fileName.matches(request.auth.uid + '__p[2-5][.]jpg'));
    }
  }
}
